문자열 -> 배열
[char]
char[] chars = my_string.toCharArray();   문자열을 문자 배열로 변환
char temp = chars[num1];                  (교환을 위해 임시로 저장)
chars[num1] = chars[num2];                (num2의 문자를 num1 자리에)
chars[num2] = temp;                       (임시 저장했던 문자를 num2 자리에)
return new String(chars);                 다시 배열을 문자열로 바꿔서 반환
[sb]
char ch1 = sb.charAt(num1);	      num1 위치 문자 추출
char ch2 = sb.charAt(num2);	      num2 위치 문자 추출
sb.setCharAt(num1, ch2);	      num1 위치에 ch2 설정
sb.setCharAt(num2, ch1);	      num2 위치에 ch1 설정
sb.toString();	                  다시 문자열로 반환


문자열 대문자 소문자
Character.isUpperCase(ch)	ch가 대문자인지 확인
Character.isLowerCase(ch)	ch가 소문자인지 확인
Character.toUpperCase(ch)	ch를 대문자로 변환
Character.toLowerCase(ch)	ch를 소문자로 변환

StringBuilder 주요 메서드
append()	문자열 끝에 붙이기
insert()	원하는 위치에 삽입
delete()	특정 부분 삭제
reverse()	문자열 뒤집기
toString()	최종 문자열로 변환

배열(붙이기)
StringBuilder sb = new StringBuilder();//홀-짝수를 붙일 문자열StringBuilder선언
for (int num : num_list) { // num_list를 반복하면서 num이라는 변수에 넣기
    if (num % 2 == 0) {
       even.append(num);  //num을 even에 붙이기(문자열로 붙이기)
    } else {
       odd.append(num);  //num을 even에 붙이기(문자열로 붙이기)

    }
    int oddNum = Integer.parseInt(odd.toString());   //붙인 문자열을 toString으로 바꾸고, Integer.parseInt()로 문자열을 정수로 변환("351" -> 351)
    int evenNum = Integer.parseInt(even.toString());
    return oddNum + evenNum;
}

더미 배열
import java.util.*;                                       //사용하기 위한 조건
List<Integer> list = new ArrayList<>();                   //생성 <성질>
list.add(숫자);                                            //값을 추가
list.size();                                              //.length와 같은 코드
list.get(인덱스)                                           //'인덱스'번째 요소 가져옴
list.set(인덱스, 새값)                                      // 인덱스 -> 새값
list.stream().mapToInt(Integer::intValue).toArray();      // Integer->Int 변환(삽입시 필요)
list.toArray(new String[0]);                              // list<String> -> String[ ]
